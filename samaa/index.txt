    <script>
        const bgMusic = document.getElementById("bgMusic");
        const musicBtn = document.getElementById("musicBtn");
        let isPlaying = false;

        function startExperience() {
            if (!isPlaying) {
                bgMusic.play().then(() => {
                    isPlaying = true;
                    musicBtn.innerText = "🎵";
                    musicBtn.classList.add("playing");
                }).catch(err => {
                    console.log("تأكد من وجود ملف الأغنية song.mp3 بجوار ملف الكود");
                });
            }
            goToPage(2);
        }

        function toggleMusic() {
            if (isPlaying) {
                bgMusic.pause();
                isPlaying = false;
                musicBtn.innerText = "🔇";
                musicBtn.classList.remove("playing");
            } else {
                bgMusic.play();
                isPlaying = true;
                musicBtn.innerText = "🎵";
                musicBtn.classList.add("playing");
            }
        }

        const startDate = new Date("2023-09-24T00:00:00"); 
        const birthdayDate = new Date("2026-11-05T00:00:00"); 
        const stopAtBirthday = false; 

        function updateTimer() {
            const now = new Date();
            const targetDate = (stopAtBirthday && now > birthdayDate) ? birthdayDate : now;
            
            let diff = targetDate - startDate;
            if (diff < 0) diff = 0;

            const days = Math.floor(diff / (1000 * 60 * 60 * 24));
            const hours = Math.floor((diff / (1000 * 60 * 60)) % 24);
            const minutes = Math.floor((diff / 1000 / 60) % 60);
            const seconds = Math.floor((diff / 1000) % 60);

            document.getElementById("days").innerText = days;
            document.getElementById("hours").innerText = hours < 10 ? '0' + hours : hours;
            document.getElementById("minutes").innerText = minutes < 10 ? '0' + minutes : minutes;
            document.getElementById("seconds").innerText = seconds < 10 ? '0' + seconds : seconds;
        }

        setInterval(updateTimer, 1000);

        const textToType = "عارفة إيه؟ أنتِ ألطف وأجمل إنسانة قابلتها في حياتي كلها! 💖";
        let charIndex = 0;
        const typewriterElement = document.getElementById("typewriter");

        function typeWriter() {
            if (charIndex < textToType.length) {
                typewriterElement.innerHTML += textToType.charAt(charIndex);
                charIndex++;
                setTimeout(typeWriter, 65);
            }
        }

        window.onload = () => {
            typeWriter();
            createFloatingElements();
            updateTimer();
        };

        function goToPage(pageNumber) {
            document.querySelectorAll('.page').forEach(page => {
                page.classList.remove('active');
            });
            document.getElementById('page' + pageNumber).classList.add('active');
            window.scrollTo(0, 0);
        }

        let currentReason = 0;
        const totalReasons = 4;

        function revealNextReason() {
            if (currentReason < totalReasons) {
                currentReason++;
                const card = document.getElementById('reason' + currentReason);
                card.classList.add('show');
                
                const counter = document.getElementById('counterText');
                counter.innerText = `رسالة ${currentReason} من ${totalReasons}`;

                const btn = document.getElementById('revealBtn');
                if (currentReason === totalReasons) {
                    btn.innerText = "شوفي خطة اليوم وذكرياتنا 💫";
                    counter.style.display = 'none';
                }
            } else {
                goToPage(3);
            }
        }

        function createFloatingElements() {
            const bg = document.getElementById('floatingBg');
            const symbols = ['🌸', '💖', '💕', '✨', '🦋', '💗'];
            
            for (let i = 0; i < 20; i++) {
                let item = document.createElement('div');
                item.className = 'float-item';
                item.innerText = symbols[Math.floor(Math.random() * symbols.length)];
                item.style.left = Math.random() * 100 + 'vw';
                item.style.animationDuration = (Math.random() * 6 + 6) + 's';
                item.style.animationDelay = (Math.random() * 5) + 's';
                item.style.fontSize = (Math.random() * 12 + 14) + 'px';
                bg.appendChild(item);
            }
        }
    </script>
